cmake_minimum_required(VERSION 3.20)
project(EcoFFT VERSION 1.0.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "The bundled Eco SDK targets Linux x86_64. Use the Docker build on other hosts.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ECO_SANITIZE "Instrument authored code and client with ASan/UBSan" OFF)
option(ECO_FFTW_REFERENCE "Build the optional external FFTW comparison client" OFF)
set(ECO_SDK "${CMAKE_SOURCE_DIR}/vendor/eco")
file(GLOB ECO_PUBLIC_DIRS LIST_DIRECTORIES true "${ECO_SDK}/*/SharedFiles")
include_directories(include ${ECO_PUBLIC_DIRS})
add_compile_definitions(ECO_LINUX ECO_X86_64)
set(ECO_SOURCES src/fft_kernel.c src/eco_dsp.c src/eco_objects.c)
function(component_flags target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-const-variable
    -ffreestanding -fno-builtin -fno-tree-loop-distribute-patterns -fno-stack-protector
    -fvisibility=hidden)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(ECO_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endfunction()
add_library(fft_objects OBJECT ${ECO_SOURCES})
component_flags(fft_objects)
foreach(kind IN ITEMS DIT DIF)
  if(kind STREQUAL "DIT")
    set(algorithm 0)
    set(cid C345FB78707144109821A6EF25310070)
  else()
    set(algorithm 1)
    set(cid C345FB78707144109821A6EF25310071)
  endif()
  add_library(eco_fft_${kind} STATIC src/eco_factory.c $<TARGET_OBJECTS:fft_objects>)
  target_compile_definitions(eco_fft_${kind} PRIVATE ECO_COMPONENT_ALGORITHM=${algorithm} ECO_LIB)
  component_flags(eco_fft_${kind})
  set_target_properties(eco_fft_${kind} PROPERTIES OUTPUT_NAME ${cid})
  add_library(eco_fft_${kind}_dynamic SHARED src/eco_factory.c $<TARGET_OBJECTS:fft_objects>)
  target_compile_definitions(eco_fft_${kind}_dynamic PRIVATE ECO_COMPONENT_ALGORITHM=${algorithm} ECO_DLL)
  component_flags(eco_fft_${kind}_dynamic)
  if(ECO_SANITIZE)
    target_link_options(eco_fft_${kind}_dynamic PRIVATE -fsanitize=address,undefined)
  else()
    target_link_options(eco_fft_${kind}_dynamic PRIVATE -nostdlib -nodefaultlibs -Wl,-z,defs -Wl,-z,noexecstack)
  endif()
  set_target_properties(eco_fft_${kind}_dynamic PROPERTIES PREFIX "" OUTPUT_NAME ${cid})
endforeach()
set(ECO_RUNTIME)
foreach(spec IN ITEMS "Eco.System1/00000000000000000000000053595333"
    "Eco.InterfaceBus1/00000000000000000000000042757331"
    "Eco.MemoryManager1/0000000000000000000000004D656D31"
    "Eco.FileSystemManagement1/00000000000000000000000046534D31")
  string(REPLACE "/" ";" parts ${spec})
  list(GET parts 0 package)
  list(GET parts 1 cid)
  list(APPEND ECO_RUNTIME "${ECO_SDK}/${package}/BuildFiles/Linux/x86_64/StaticRelease/lib${cid}.a")
endforeach()
set(ECO_CLIENT_SOURCES client/eco_main.c client/transform.c client/checks.c client/measure.c)
add_executable(eco_fft ${ECO_CLIENT_SOURCES} client/eco_runtime.c)
add_executable(eco_fft_static ${ECO_CLIENT_SOURCES})
target_compile_definitions(eco_fft_static PRIVATE ECO_LIB ECO_STATIC_CLIENT)
target_link_libraries(eco_fft_static PRIVATE eco_fft_DIT eco_fft_DIF)
add_executable(eco_fft_faults client/fault_main.c client/faults.c)
target_compile_definitions(eco_fft_faults PRIVATE ECO_LIB)
target_include_directories(eco_fft_faults PRIVATE src)
target_link_libraries(eco_fft_faults PRIVATE eco_fft_DIT eco_fft_DIF)
add_executable(eco_fft_example examples/use_component.c)
add_executable(eco_fft_acom examples/check_acom.c)
set(ECO_APPS eco_fft eco_fft_static eco_fft_faults eco_fft_example eco_fft_acom)
set(ECO_DYNAMIC_APPS eco_fft eco_fft_example eco_fft_acom)
if(ECO_FFTW_REFERENCE)
  find_path(FFTW_INCLUDE_DIR fftw3.h REQUIRED)
  find_library(FFTW_LONG_DOUBLE_LIBRARY fftw3l REQUIRED)
  add_executable(eco_fft_reference client/compare_fftw.c client/eco_runtime.c)
  target_include_directories(eco_fft_reference PRIVATE ${FFTW_INCLUDE_DIR})
  target_link_libraries(eco_fft_reference PRIVATE ${FFTW_LONG_DOUBLE_LIBRARY})
  list(APPEND ECO_APPS eco_fft_reference)
  list(APPEND ECO_DYNAMIC_APPS eco_fft_reference)
endif()
foreach(app IN LISTS ECO_APPS)
  target_compile_options(${app} PRIVATE -Wall -Wextra -Werror -Wno-unused-const-variable)
  # Dynamic clients must never link the FFT implementation or factory objects.
  target_link_libraries(${app} PRIVATE ${ECO_RUNTIME} m dl)
  if(ECO_SANITIZE)
    target_compile_options(${app} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${app} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
foreach(app IN LISTS ECO_DYNAMIC_APPS)
  add_dependencies(${app} eco_fft_DIT_dynamic eco_fft_DIF_dynamic)
endforeach()
